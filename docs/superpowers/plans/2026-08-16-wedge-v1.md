# Wedge v1 Implementation Plan

> **Historical:** This plan drove the initial v1 build (popover, Force-click, header Done). **Canonical product truth is now** [`docs/product/phases/v1.md`](../../product/phases/v1.md) and `.cursor/rules/wedge-product.mdc` (floating panel, no Force-click, no panel Done button, Settings via ⌘,/menu, Notion-style drag grips). Do not re-implement from this plan without reconciling those docs.

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship Wedge v1 as a macOS menu-bar-only app: ordered one thing / Up next list, Done/pop with sound, local persistence, launch at login, and a 40-character hard limit.

**Architecture:** Keep domain logic in a pure `@Observable` `WedgeStore` (testable without UI). Persist via a small `WedgePersisting` adapter (UserDefaults JSON, injectable for tests). Drive the menu bar with AppKit `NSStatusItem` + `NSPopover` hosting SwiftUI `PopoverContentView` so left/right/⌥/Force-click can be handled correctly. App runs as accessory (`LSUIElement`); no `WindowGroup`.

**Tech Stack:** Swift 5 / SwiftUI + AppKit on macOS, Swift Testing (`Testing`), UserDefaults persistence, `SMAppService` for login item, `NSSound` for pop, Xcode file-system-synchronized `wedge/` and `wedgeTests/` folders (new files auto-join targets).

## Global Constraints

- Product spec: `docs/product/phases/v1.md` (also overview/roadmap).
- UI copy: **one thing**, **Up next**, **Done** only — never push/pop/stack/P0 in UI.
- Menu-bar only: `LSUIElement` / accessory activation; no Dock icon.
- Max item text length: **40** characters; reject longer input; **no** truncation/ellipsis system.
- Adds always append to the **bottom** of Up next (first add on empty list becomes the one thing by being index 0).
- Done no-ops when the list is empty.
- Out of v1: global hotkeys, Preferences UI, cloud/sync, Dock mode, confetti.
- App Sandbox is already enabled (`ENABLE_APP_SANDBOX = YES`) — keep it; use container-safe APIs.
- Unit tests: Swift Testing in `wedgeTests`. Prefer running unit tests only while iterating (`-only-testing:wedgeTests`) to avoid UI-test app flashing.
- Align docs/rules if behavior drifts (`wedge-keep-aligned`).

```swift
import Foundation
import Observation

enum WedgeStoreError: Error, Equatable {
    case emptyText
    case textTooLong(max: Int)
}

@Observable
final class WedgeStore {
    static let maxTextLength = 40

    private(set) var items: [WedgeItem]

    var oneThing: WedgeItem? { items.first }

    init(items: [WedgeItem] = []) {
        self.items = items
    }

    func replaceAll(_ items: [WedgeItem]) {
        self.items = items
    }

    @discardableResult
    func add(_ rawText: String) throws -> WedgeItem {
        let text = try Self.validatedText(rawText)
        let item = WedgeItem(text: text)
        items.append(item)
        return item
    }

    func update(id: UUID, text rawText: String) throws {
        let text = try Self.validatedText(rawText)
        guard let index = items.firstIndex(where: { $0.id == id }) else { return }
        items[index].text = text
    }

    func delete(id: UUID) {
        items.removeAll { $0.id == id }
    }

    func move(from offsets: IndexSet, to offset: Int) {
        items.move(fromOffsets: offsets, toOffset: offset)
    }

    @discardableResult
    func completeOneThing() -> WedgeItem? {
        guard !items.isEmpty else { return nil }
        return items.removeFirst()
    }

    private static func validatedText(_ rawText: String) throws -> String {
        let text = rawText.trimmingCharacters(in: .whitespacesAndNewlines)
        guard !text.isEmpty else { throw WedgeStoreError.emptyText }
        guard text.count <= maxTextLength else {
            throw WedgeStoreError.textTooLong(max: maxTextLength)
        }
        return text
    }
}
```

`wedge/Persistence/WedgePersisting.swift`:

```swift
import Foundation

protocol WedgePersisting: Sendable {
    func load() throws -> [WedgeItem]
    func save(_ items: [WedgeItem]) throws
}

struct UserDefaultsWedgePersistence: WedgePersisting {
    private let defaults: UserDefaults
    private let key: String

    init(defaults: UserDefaults = .standard, key: String = "wedge.items") {
        self.defaults = defaults
        self.key = key
    }

    func load() throws -> [WedgeItem] {
        guard let data = defaults.data(forKey: key) else { return [] }
        return try JSONDecoder().decode([WedgeItem].self, from: data)
    }

    func save(_ items: [WedgeItem]) throws {
        let data = try JSONEncoder().encode(items)
        defaults.set(data, forKey: key)
    }
}
```

```swift
import AppKit
import SwiftUI

@MainActor
final class StatusItemController: NSObject {
    private let store: WedgeStore
    private let persistence: any WedgePersisting
    private var statusItem: NSStatusItem?
    private var popover: NSPopover?
    private var observationTask: Task<Void, Never>?

    init(store: WedgeStore, persistence: any WedgePersisting) {
        self.store = store
        self.persistence = persistence
    }

    func start() {
        let item = NSStatusBar.system.statusItem(withLength: NSStatusItem.variableLength)
        if let button = item.button {
            button.image = NSImage(systemSymbolName: "wedge.fill", accessibilityDescription: "Wedge")
                ?? NSImage(systemSymbolName: "circle.fill", accessibilityDescription: "Wedge")
            button.imagePosition = .imageLeft
            button.target = self
            button.action = #selector(statusItemPrimaryAction(_:))
            button.sendAction(on: [.leftMouseUp, .rightMouseUp])
        }
        statusItem = item
        refreshTitle()
        observationTask = Task { [weak self] in
            // Polling observation is OK for v1; or use withObservationTracking loop
            while !Task.isCancelled {
                await MainActor.run { self?.refreshTitle() }
                try? await Task.sleep(for: .milliseconds(200))
            }
        }
        // Better: call refreshTitle() after every store mutation from UI/controller instead of polling.
        // Prefer explicit `persistAndRefresh()` after mutations; remove polling once wired.
    }

    func refreshTitle() {
        guard let button = statusItem?.button else { return }
        if let text = store.oneThing?.text {
            button.title = text
        } else {
            button.title = ""
        }
    }

    func persistAndRefresh() {
        try? persistence.save(store.items)
        refreshTitle()
    }

    @objc private func statusItemPrimaryAction(_ sender: NSStatusBarButton) {
        guard let event = NSApp.currentEvent else { return }
        if event.type == .rightMouseUp {
            showContextMenu()
            return
        }
        if event.modifierFlags.contains(.option) {
            performDone()
            return
        }
        togglePopover()
    }

    private func showContextMenu() {
        let menu = NSMenu()
        let doneItem = NSMenuItem(title: "Done", action: #selector(menuDone(_:)), keyEquivalent: "")
        doneItem.target = self
        doneItem.isEnabled = store.oneThing != nil
        menu.addItem(doneItem)
        menu.addItem(.separator())
        menu.addItem(NSMenuItem(title: "Quit Wedge", action: #selector(NSApplication.terminate(_:)), keyEquivalent: "q"))
        statusItem?.menu = menu
        statusItem?.button?.performClick(nil)
        statusItem?.menu = nil
    }

    @objc private func menuDone(_ sender: Any?) {
        performDone()
    }

    func performDone() {
        guard store.completeOneThing() != nil else { return }
        // Sound + animation wired in Task 6
        persistAndRefresh()
        popover?.contentViewController?.view.needsDisplay = true
    }

    private func togglePopover() {
        // Task 5 fills real content; use empty NSView for now
        if popover == nil {
            let pop = NSPopover()
            pop.behavior = .transient
            pop.contentSize = NSSize(width: 280, height: 320)
            pop.contentViewController = NSHostingController(
                rootView: PopoverContentView(store: store, onChange: { [weak self] in
                    self?.persistAndRefresh()
                })
            )
            self.popover = pop
        }
        guard let button = statusItem?.button, let popover else { return }
        if popover.isShown {
            popover.performClose(nil)
        } else {
            popover.show(relativeTo: button.bounds, of: button, preferredEdge: .minY)
        }
    }
}
```

```swift
import SwiftUI

struct PopoverContentView: View {
    @Bindable var store: WedgeStore
    var onChange: () -> Void
    @State private var draft = ""
    @State private var selection: UUID?
    @State private var donePulse = false

    private var addPlaceholder: String {
        store.items.isEmpty ? "What's your one thing?" : "Add to up next…"
    }

    var body: some View {
        VStack(alignment: .leading, spacing: 8) {
            HStack {
                Text("Wedge")
                    .font(.headline)
                Spacer()
                Button("Done") { performDone() }
                    .disabled(store.oneThing == nil)
                    .keyboardShortcut(.return, modifiers: .command)
            }

            List(selection: $selection) {
                ForEach(store.items) { item in
                    TextField(
                        "Item",
                        text: Binding(
                            get: { item.text },
                            set: { newValue in
                                try? store.update(id: item.id, text: newValue)
                                onChange()
                            }
                        )
                    )
                    .font(item.id == store.oneThing?.id ? .body.weight(.semibold) : .body)
                    .tag(item.id)
                }
                .onMove { indices, newOffset in
                    store.move(from: indices, to: newOffset)
                    onChange()
                }
                .onDelete { indexSet in
                    for index in indexSet {
                        store.delete(id: store.items[index].id)
                    }
                    onChange()
                }
            }
            .listStyle(.inset(alternatesRowBackgrounds: true))
            .frame(minHeight: 180)
            .onDeleteCommand {
                if let selection {
                    store.delete(id: selection)
                    self.selection = nil
                    onChange()
                }
            }

            TextField(addPlaceholder, text: $draft)
                .textFieldStyle(.roundedBorder)
                .onSubmit { submitDraft() }
        }
        .padding(12)
        .frame(width: 280)
        .opacity(donePulse ? 0.85 : 1)
    }

    private func submitDraft() {
        do {
            _ = try store.add(draft)
            draft = ""
            onChange()
        } catch {
            NSSound.beep()
        }
    }

    private func performDone() {
        guard store.completeOneThing() != nil else { return }
        withAnimation(.easeOut(duration: 0.15)) { donePulse = true }
        DispatchQueue.main.asyncAfter(deadline: .now() + 0.15) { donePulse = false }
        onChange()
```
